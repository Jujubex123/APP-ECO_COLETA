  <div class="card">
    <h2>Agendar Coleta de Lixo</h2>

    <label for="data">Escolha a Data:</label>
    <input type="date" id="data">
    <button id="btnData" class="btn" onclick="salvarData()">Salvar Data</button>

    <label for="hora">Escolha o Horário:</label>
    <input type="time" id="hora">
    <button id="btnHora" class="btn" onclick="salvarHora()">Salvar Horário</button>

    <button id="btnAgendar" class="btn" onclick="agendar()">Confirmar Agendamento</button>
    <button id="btnCancelar" class="btn" onclick="cancelar()">Cancelar Agendamento</button>

    <p id="mensagem"></p>
  </div>

  <script>
    let dataSelecionada = localStorage.getItem('dataColeta') || '';
    let horaSelecionada = localStorage.getItem('horaColeta') || '';

    function salvarData() {
      const data = document.getElementById('data').value;
      const msg = document.getElementById('mensagem');
      if (!data) {
        msg.textContent = "⚠️ Selecione uma data válida.";
        msg.style.color = "var(--cor-erro)";
        return;
      }
      localStorage.setItem('dataColeta', data);
      dataSelecionada = data;
      msg.textContent = `📅 Data salva: ${new Date(data).toLocaleDateString('pt-BR')}`;
      msg.style.color = "var(--verde-primario)";
    }

    function salvarHora() {
      const hora = document.getElementById('hora').value;
      const msg = document.getElementById('mensagem');
      if (!hora) {
        msg.textContent = "⚠️ Selecione um horário válido.";
        msg.style.color = "var(--cor-erro)";
        return;
      }
      localStorage.setItem('horaColeta', hora);
      horaSelecionada = hora;
      msg.textContent = `🕒 Horário salvo: ${hora}`;
      msg.style.color = "var(--azul-primario)";
    }

    function agendar() {
      const msg = document.getElementById('mensagem');
      if (!dataSelecionada || !horaSelecionada) {
        msg.textContent = "⚠️ Salve a data e o horário antes de confirmar.";
        msg.style.color = "var(--cor-erro)";
        return;
      }
      const dataHora = `${dataSelecionada}T${horaSelecionada}`;
      localStorage.setItem('agendamentoColeta', dataHora);
      msg.textContent = `✅ Coleta agendada para: ${new Date(dataHora).toLocaleString('pt-BR')}`;
      msg.style.color = "var(--azul-primario)";
    }

    function cancelar() {
      localStorage.removeItem('agendamentoColeta');
      localStorage.removeItem('dataColeta');
      localStorage.removeItem('horaColeta');
      document.getElementById('mensagem').textContent = "❌ Agendamento cancelado.";
      document.getElementById('mensagem').style.color = "var(--laranja-secundario)";
    }

    window.onload = function() {
      const dataSalva = localStorage.getItem('dataColeta');
      const horaSalva = localStorage.getItem('horaColeta');
      const agendamentoSalvo = localStorage.getItem('agendamentoColeta');
      const msg = document.getElementById('mensagem');

      if (dataSalva) document.getElementById('data').value = dataSalva;
      if (horaSalva) document.getElementById('hora').value = horaSalva;

      if (agendamentoSalvo) {
        const dataFormatada = new Date(agendamentoSalvo).toLocaleString('pt-BR');
        msg.textContent = `📦 Coleta já agendada para: ${dataFormatada}`;
      }
    }
  </script>
