<script>
  // Solicita permissão de notificação
  async function pedirPermissao() {
    if (Notification.permission !== "granted") {
      await Notification.requestPermission();
    }
  }

  pedirPermissao();

  // Função principal do botão
  document.getElementById("btnNotificar").addEventListener("click", () => {
    const dataInput = document.getElementById("dataColeta").value;
    if (!dataInput) {
      alert("Por favor, selecione o dia da coleta!");
      return;
    }

    const dataColeta = new Date(dataInput);
    const hoje = new Date();

    if (dataColeta <= hoje) {
      alert("Escolha uma data futura!");
      return;
    }

    // Exibe notificação imediata (simulação)
    new Notification("Lembrete de Coleta", {
      body: `Sua coleta está agendada para ${dataColeta.toLocaleDateString("pt-BR")}`,
      icon: "https://cdn-icons-png.flaticon.com/512/190/190411.png"
    });

    alert(`Notificação programada para ${dataColeta.toLocaleDateString("pt-BR")}`);
  });
</script>
